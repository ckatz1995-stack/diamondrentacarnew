function _has(obj, key) {

function _pickKey(sample, candidates, fallback) {
  for (const k of candidates) {

  }
  return fallback;
}

export function isOpaqueCategoryId(value) {
  const s = String(value || '').trim();
  return /^[0-9a-f]{8}-[0-9a-f-]{12,}$/i.test(s) || (/^[0-9A-Za-z]{8,}$/i.test(s) && !/^[A-Z]{1,4}$/i.test(s));
}
export function catCode(v) {
  if (!v) return "";
  let s = v;
  if (Array.isArray(s)) s = s[0];
  if (typeof s === "object") {
    // Wix reference/object may expose different keys depending on include()
    s = s.category || s.code || s.cat || s.title || s.name || s.label || s.value || s._id || "";
  }
  s = String(s || "").trim();
  if (!s || isOpaqueCategoryId(s)) return "";
  // If it's like "A - Hyundai i10" or "A/..." => take first token
  const m = s.match(/^([A-Za-z0-9Α-Ω]{1,4})/);
  const first = m ? m[1] : s;
  return String(first).replace("Χ","X").toUpperCase();
}

export async function getFleetCalendarData({ authToken, from, to } = {}) {
  await requireStaffAccess({ sessionToken: authToken, area: "fleet", action: "View" });
  await ensureFieldMapping();

  const now = new Date();
  const f = toDate(from) || new Date(new Date(now).setHours(0,0,0,0));
  const t = toDate(to) || new Date(now.getTime() + 14 * 86400000);

  const fleet = await fetchFleet();
  const bookings = await fetchBookingsInRange(iso(f), iso(t));

  // The vehicle-category catalog used to be fetched here and passed to buildGroups,
  // but buildGroups takes only `fleet` — the second argument was silently discarded,
  // so the query ran on every calendar load and its result was never read. Removed.
  // Groups are therefore labelled from each vehicle's own category code rather than
  // the catalog title. Wiring the titles in is a UI decision; the fetcher that used to
  // supply them is in git history if someone takes it on.
  const groups = buildGroups(fleet);
  const { items, unassigned } = buildItems(bookings);
  const heat = buildHeatmap({ from: f, to: t, fleet, items });

  return { groups, items, unassigned, heat };
}
 
/**
 * Move booking to another vehicle ONLY.
 * Accept BOTH param names: newVehicleId OR resourceId (backwards compatible).
 * @param {any} [options]
 */
export async function moveBookingVehicleOnly({ authToken, bookingId, newVehicleId, resourceId } = {}) {
  await requireStaffAccess({ sessionToken: authToken, area: "fleet", action: "Edit" });
  await ensureFieldMapping();

  const rawTarget = newVehicleId || resourceId;
  if (!bookingId || rawTarget === undefined || rawTarget === null || rawTarget === "") return { success: false, message: "MissingParams" };

  const b = await wixData.get(COLLECTIONS.Bookings, bookingId, { suppressAuth: true });
  if (!b) return { success: false, message: "NotFound" };
  const rs = String(b[F.rentalState] || "");
  if (rs === "Active Rental" || rs === "Pre Check-In") {
    return { success: false, message: "LockedRentalState" };
  }

  if (isNonRevenueBooking(b)) return { success: false, message: "NonRevenueLocked" };

  const start = toDate(b?.[F.pickupDT]);
  const end = toDate(b?.[F.dropoffDT]);
  if (!start || !end) return { success: false, message: "MissingDates" };

  const targetVehicleId = refId(rawTarget);

  if (targetVehicleId) {
    const res = await wixData.query(COLLECTIONS.Bookings)
      .eq(F.assignedFleetId, String(targetVehicleId))
      .ne(F.bookingId, bookingId)
      .ne(F.status, "Canceled")
      // Same overlap rule as bookingsOverlap() in backend/availability.js, in DB form.
      .lt(F.pickupDT, iso(end))
      .gt(F.dropoffDT, iso(start))
      .limit(1000)
      .find({ suppressAuth: true });

    if ((res.items || []).length > 0) return { success: false, message: "Overlap" };
    b[F.assignedFleetId] = String(targetVehicleId);
    const meta = await getFleetMetaById(targetVehicleId);
    if (meta) {
      b.assignedVehiclePlate = meta.plate || "";
      b.assignedVehicleModel = meta.model || "";
      b.assignedVehicleLabel = meta.label || "";
      b.assignedCategoryCode = meta.assignedCategoryCode || "";
    }
  } else {
    b[F.assignedFleetId] = null;
    b.assignedVehiclePlate = "";
    b.assignedVehicleModel = "";
    b.assignedVehicleLabel = "";
    b.assignedCategoryCode = "";
  }

  await wixData.update(COLLECTIONS.Bookings, b, { suppressAuth: true });
  return { success: true, assignedVehicle: targetVehicleId || null };
}
 
/**
 * Confirm + auto-assign.
 * (Safe default: sample algorithm)
  * @param {{ authToken?: any, bookingId?: any }} [options]
 */
export async function confirmAndAutoAssign({ authToken, bookingId } = {}) {
  await requireStaffAccess({ sessionToken: authToken, area: "fleet", action: "Edit" });
  await ensureFieldMapping();

  if (!bookingId) return { success: false, message: "MissingParams" };

  const b = await wixData.get(COLLECTIONS.Bookings, bookingId, { suppressAuth: true });
  if (!b) return { success: false, message: "NotFound" };

  // Only transition Pending/empty bookings to Confirmed; never overwrite Canceled or active/closed rentals.
  // Normalize for comparison so legacy lowercase/whitespace values (e.g. "pending", "Confirmed ") still flow.
  const currentStatusRaw = String(b[F.status] || "").trim();
  const currentStatusNorm = currentStatusRaw.toLowerCase();
  if (currentStatusNorm === "canceled" || currentStatusNorm === "cancelled") {
    return { success: false, message: "Cannot confirm a canceled booking" };
  }
  if (currentStatusNorm && currentStatusNorm !== "pending" && currentStatusNorm !== "confirmed") {
    return { success: false, message: `Cannot confirm a booking in status: ${currentStatusRaw}` };
  }
  if (currentStatusNorm !== "confirmed") {
    b[F.status] = "Confirmed";
  }
// If already assigned
  if (refId(b?.[F.assignedFleetId])) return { success: true, assigned: true, alreadyAssigned: true };

  const start = toDate(b?.[F.pickupDT]);
  const end = toDate(b?.[F.dropoffDT]);
  const cat = catCode(b?.[F.categoryId]) || catCode(b?.[F.category]) || "";

  if (!start || !end) return { success: false, message: "MissingDates" };

  const fleet = await fetchFleet();
  const candidates = (fleet || []).filter(v => {
    const vcat = catCode(v?.[VF.category]);
    return !cat || vcat === cat || catLabel(v?.[VF.category]) === cat;
  });

  for (const v of candidates) {
    const check = await wixData.query(COLLECTIONS.Bookings)
      .eq(F.assignedFleetId, String(v[VF.id]))
      .ne(F.status, "Canceled")
      // Same overlap rule as bookingsOverlap() in backend/availability.js, in DB form.
      .lt(F.pickupDT, iso(end))
      .gt(F.dropoffDT, iso(start))
      .limit(1)
      .find({ suppressAuth: true });

    if ((check.items || []).length === 0) {
      b[F.assignedFleetId] = String(v[VF.id]);
      const meta = await getFleetMetaById(v[VF.id]);
      if (meta) {
        b.assignedVehiclePlate = meta.plate || "";
        b.assignedVehicleModel = meta.model || "";
        b.assignedVehicleLabel = meta.label || "";
        b.assignedCategoryCode = meta.assignedCategoryCode || "";
      }
      await wixData.update(COLLECTIONS.Bookings, b, { suppressAuth: true });
      return { success: true, assigned: true, vehicleId: String(v[VF.id]) };
    }
  }

  return { success: true, assigned: false };
}

function buildGroups(fleet) {
  const byCat = new Map();

  for (const v of (fleet || [])) {
    const cat = fleetCatCode(v) || "Other";
    if (!byCat.has(cat)) byCat.set(cat, []);
    byCat.get(cat).push(v);
  }

  const cats = Array.from(byCat.keys()).sort((a, b) => {
    const ax = a === "Other" ? "ZZZ" : a;
    const bx = b === "Other" ? "ZZZ" : b;
    return ax.localeCompare(bx, "el");
  });

  const groups = [];

  for (const cat of cats) {
    const parentId = `cat_${cat}`;
    const vehicles = (byCat.get(cat) || []).slice().sort((x, y) => {
      const px = String(x?.[VF.plate] || "");
      const py = String(y?.[VF.plate] || "");
      return px.localeCompare(py, "el");
    });

    // Parent category group
    groups.push({
      id: parentId,
      content: renderCategoryHeader(cat, vehicles.length),
      nestedGroups: vehicles.map(v => v[VF.id]),
      showNested: true,
      className: "fleetCatGroup"
    });

    for (const [idx, v] of vehicles.entries()) {
      const plate = String(v?.[VF.plate] || v[VF.id] || "").trim();
      const model = String(v?.[VF.model] || "").trim();
      const station = String(v?.[VF.station] || "").trim();
      const stationCode = String(v?.[VF.currentStationCode] || "").trim();
      const stationLabel = String(v?.[VF.currentStationLabel] || "").trim();
      const mileage = String(v?.[VF.mileage] || "").trim();
      const ready = yesNo(v?.[VF.ready]);
      const stall = String(v?.[VF.stall] || "").trim();
      const operationalStatus = String(v?.[VF.operationalStatus] || "").trim();
      const hardHold = v?.[VF.hardHold] === true || String(v?.[VF.hardHold] || "").toLowerCase() === "true";
      const serviceBlocked = ["service","blocked","inactive"].includes(String(operationalStatus || "").toLowerCase());
      const assignable = !hardHold && !serviceBlocked && ready !== "Not ready";
      const label = model ? `${plate} • ${model}` : plate;
      const titleLines = [
        label,
        cat ? `Category: ${cat}` : "",
        stationLabel || stationCode || station ? `Station: ${stationLabel || stationCode || station}` : "",
        mileage ? `Mileage: ${mileage}` : "",
        ready && ready !== "—" ? `Ready: ${ready}` : "",
        stall ? `Stall: ${stall}` : "",
        operationalStatus ? `Operational: ${operationalStatus}` : "",
        hardHold ? "Hard hold: Yes" : "",
        `Assignable: ${assignable ? "Yes" : "No"}`
      ].filter(Boolean);
      groups.push({
        id: v[VF.id],
        content: renderVehicleRow(v, idx, cat),
        plate,
        model,
        station,
        mileage,
        ready,
        stall,
        operationalStatus,
        hardHold,
        serviceBlocked,
        assignable,
        stationCode,
        stationLabel,
        categoryCode: cat,
        title: titleLines.join("\n"),
        className: idx % 2 === 0 ? "fleetRowEven" : "fleetRowOdd"
      });
    }
  }

  // Unassigned pseudo-category
  groups.push({
    id: "cat_Unassigned",
    content: renderCategoryHeader("Unassigned", 1),
    nestedGroups: ["unassigned"],
    showNested: true,
    className: "fleetCatGroup"
  });
  groups.push({ id: "unassigned", content: `<div class="fleetGroupCell"><span class="fleetCol fleetNo">—</span><span class="fleetCol fleetPlate">Unassigned</span><span class="fleetCol fleetModel">Queue</span><span class="fleetCol fleetStation">—</span><span class="fleetCol fleetCategory">—</span></div>`, className: "fleetRowEven" });

  return groups;
}

/**
 * Simple heatmap example (kept from your original logic style)
 */
export function buildHeatmap({ from, to, fleet, items }) {
  const start = new Date(from);
  const end = new Date(to);
  const days = Math.max(1, Math.ceil((end.getTime() - start.getTime()) / 86400000));
  const fleetIds = new Set((fleet || []).map(v => String(v?.[VF.id] || "")));
 
  // heat[vehicleId][dayIndex] = count
  const heat = {};
  for (const vid of fleetIds) heat[vid] = Array(days).fill(0);
 
  for (const it of (items || [])) {
    const vid = String(it.resource || "");
    if (!vid || !heat[vid]) continue;
 
    const s = new Date(it.start);
    const e = new Date(it.end);
 
    const sIdx = Math.max(0, Math.floor((s.getTime() - start.getTime()) / 86400000));
    const eIdx = Math.min(days - 1, Math.floor((e.getTime() - start.getTime()) / 86400000));
 
    for (let i = sIdx; i <= eIdx; i++) heat[vid][i] += 1;
  }
 
  return { from: start.toISOString(), to: end.toISOString(), days, heat };
}
