function parseDateOnly(value){
  const raw = text(value, '');
  if (!raw) return null;
  const dt = new Date(raw);
  if (Number.isNaN(dt.getTime())) return null;
  dt.setHours(0,0,0,0);
  return dt;
}
function seasonMatchesDate(date, season){
  if (!(date instanceof Date) || Number.isNaN(date.getTime())) return false;
  const start = parseDateOnly(season?.startDate);
  const end = parseDateOnly(season?.endDate);
  if (!start || !end) return false;
  const check = new Date(date);
  check.setHours(0,0,0,0);
  if (season?.repeatYearly) {
    const monthDay = (d) => ({ m: d.getUTCMonth() + 1, day: d.getUTCDate() });
    const s = monthDay(start);
    const e = monthDay(end);
    const c = monthDay(check);
    const cv = c.m * 100 + c.day;
    const sv = s.m * 100 + s.day;
    const ev = e.m * 100 + e.day;
    return sv <= ev ? (cv >= sv && cv <= ev) : (cv >= sv || cv <= ev);
  }
  return check >= start && check <= end;
}
function findActiveSeason(pricingSeasons = [], pickupDateTime){
  const rows = Array.isArray(pricingSeasons) ? pricingSeasons.filter((row) => row && row.active !== false) : [];
  const matches = rows.filter((row) => seasonMatchesDate(pickupDateTime, row));
  matches.sort((a, b) => {
    const pa = toNumber(b?.priority, 0) - toNumber(a?.priority, 0);
    if (pa !== 0) return pa;
    return sortByOrder(a, b);
  });
  return matches[0] || null;
}
 
function mergeRowsByKey(defaultRows, cmsRows){
  const map = new Map();
  defaultRows.forEach((row) => { const key = normalizeKey(row?.key); if (key) map.set(key, { ...row, key }); });
  cmsRows.forEach((row) => {
    const key = normalizeKey(row?.key);
    if (!key) return;
    const base = map.get(key) || {};
    const merged = { ...base, ...row, key, _id: row._id || base._id || '' };
    map.set(key, merged);
  });

export function resolveDynamicPricingRate({ categoryCode = '', vehicleCategoryId = '', billableDays = 1, pickupDateTime = null, dropoffDateTime = null, catalog = null, fallbackPricePerDay = 0 } = {}) {
  void dropoffDateTime;
  const safeCatalog = catalog || {};
  const seasons = Array.isArray(safeCatalog.pricingSeasons) ? safeCatalog.pricingSeasons : [];
  const rules = Array.isArray(safeCatalog.categoryRateRules) ? safeCatalog.categoryRateRules : [];
  const resolvedSeason = findActiveSeason(seasons, pickupDateTime);
  const code = text(categoryCode, '').toUpperCase();
  const categoryId = text(vehicleCategoryId, '');
  const days = Math.max(1, Math.floor(toNumber(billableDays, 1)));
  const candidates = rules.filter((rule) => {
    if (!rule || rule.active === false) return false;
    if (code && text(rule.categoryCode, '').toUpperCase() && text(rule.categoryCode, '').toUpperCase() !== code) return false;
    if (!code && categoryId && text(rule.vehicleCategoryId, '') && text(rule.vehicleCategoryId, '') !== categoryId) return false;
    const minDays = Math.max(1, Math.floor(toNumber(rule.minDays, 1)));
    const maxDays = Math.max(0, Math.floor(toNumber(rule.maxDays, 0)));
    if (days < minDays) return false;
    if (maxDays > 0 && days > maxDays) return false;
    const ruleSeasonKey = normalizeKey(rule.seasonKey || '');
    if (resolvedSeason?.key) return !ruleSeasonKey || ruleSeasonKey === normalizeKey(resolvedSeason.key);
    return !ruleSeasonKey;
  });
  candidates.sort((a, b) => {
    const seasonMatchDiff = Number(normalizeKey(a?.seasonKey || '') === normalizeKey(resolvedSeason?.key || '')) - Number(normalizeKey(b?.seasonKey || '') === normalizeKey(resolvedSeason?.key || ''));
    if (seasonMatchDiff !== 0) return -seasonMatchDiff;
    const minA = Math.max(1, Math.floor(toNumber(a?.minDays, 1)));
    const minB = Math.max(1, Math.floor(toNumber(b?.minDays, 1)));
    if (minA !== minB) return minB - minA;
    const rangeA = Math.max(0, Math.floor(toNumber(a?.maxDays, 0))) || 999999;
    const rangeB = Math.max(0, Math.floor(toNumber(b?.maxDays, 0))) || 999999;
    const widthA = rangeA - minA;
    const widthB = rangeB - minB;
    if (widthA !== widthB) return widthA - widthB;
    return sortByOrder(a, b);
  });
  const rule = candidates[0] || null;
  const pricePerDay = rule ? toNumber(rule.pricePerDay, fallbackPricePerDay) : toNumber(fallbackPricePerDay, 0);
  return {
    pricePerDay,
    source: rule ? 'dynamicRule' : 'vehicleBasePrice',
    season: resolvedSeason ? { key: resolvedSeason.key, label: resolvedSeason.label, startDate: resolvedSeason.startDate, endDate: resolvedSeason.endDate } : null,
    rule: rule ? { key: rule.key, label: rule.label, minDays: rule.minDays, maxDays: rule.maxDays, seasonKey: rule.seasonKey, categoryCode: rule.categoryCode || code || '', pricePerDay: toNumber(rule.pricePerDay, 0) } : null
  };
}
