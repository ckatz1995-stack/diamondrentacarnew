/**
 * @param {any} [options]
 */
export async function loginStaff({ email, password, remember = false, userAgent = '' } = {}) {
  await bootstrapRoles();
  const safeEmail = normalizeEmail(email);
  const rawPassword = String(password || '');
  if (!safeEmail || !rawPassword) throw new Error('Συμπλήρωσε email και password.');
 
  const user = await getStaffUser(safeEmail);
  if (!user || !readBool(user.active, true)) {
    await logAudit({ entityId: safeEmail, action: 'loginFailed', summary: 'Unknown or inactive user', actorEmail: safeEmail, actorName: safeEmail });
    throw new Error('Λάθος email ή password.');
  }
 
  const role = await getRole(user.roleKey);
  if (!role || !readBool(role.active, true)) {
    await logAudit({ entityId: safeEmail, action: 'loginFailed', summary: 'Inactive role', actorEmail: safeEmail, actorName: text(user.fullName) });
    throw new Error('Λάθος email ή password.');
  }
 
  let credential = await getCredential(safeEmail);
  if (!credential) {
    const bootstrapAllowed = await canUseBootstrapLogin(user);
    const bootstrapPassword = await getBootstrapPassword();
    const bootstrapValid = !!bootstrapPassword && safeEqual(sha256(rawPassword), sha256(bootstrapPassword));
    if (!bootstrapAllowed || !bootstrapValid) {
      await logAudit({ entityId: safeEmail, action: 'loginFailed', summary: 'Invalid bootstrap login', actorEmail: safeEmail, actorName: text(user.fullName) });
      throw new Error('Λάθος email ή password.');
    }
    credential = await upsertCredential(safeEmail, rawPassword, {
      mustChangePassword: true,
      active: true,
      notes: 'Bootstrap password created automatically during first setup.'
    });
  }
 
  if (!readBool(credential.active, true)) {
    await logAudit({ entityId: safeEmail, action: 'loginFailed', summary: 'Inactive credentials', actorEmail: safeEmail, actorName: text(user.fullName) });
    throw new Error('Λάθος email ή password.');
  }
  if (!verifyPassword(rawPassword, credential)) {
    await logAudit({ entityId: safeEmail, action: 'loginFailed', summary: 'Wrong password', actorEmail: safeEmail, actorName: text(user.fullName) });
    throw new Error('Λάθος email ή password.');
  }
 
  const created = await createSession({
    email: safeEmail,
    roleKey: role.key,
    fullName: user.fullName,
    remember,
    userAgent
  });
 
  await logAudit({ entityId: safeEmail, action: 'loginSuccess', summary: 'Backroom login', actorEmail: safeEmail, actorName: text(user.fullName) });
 
  return {
    success: true,
    sessionToken: created.sessionToken,
    expiresAt: created.expiresAt,
    bootstrapUsed: !credential?._id,
    profile: buildProfile(user, role, { ...created, remember }, credential)
  };
}
 
/**
 * @param {any} [options]
 */
export async function getSessionState({ sessionToken, touch = true } = {}) {
  const rawToken = text(sessionToken);
  if (!rawToken) return { authenticated: false };
 
  const session = await getSessionByToken(rawToken);
  if (!session || !readBool(session.active, true)) return { authenticated: false };
 
  const expiresAt = session?.expiresAt ? new Date(session.expiresAt) : null;
  if (!expiresAt || Number.isNaN(expiresAt.getTime()) || expiresAt.getTime() <= Date.now()) {
    try { await wixData.update(COLLECTIONS.sessions, { ...session, active: false, revokedAt: new Date() }, { suppressAuth: true }); } catch (_) {}
    return { authenticated: false };
  }
 
  const user = await getStaffUser(session.email);
  if (!user || !readBool(user.active, true)) return { authenticated: false };
  const role = await getRole(user.roleKey);
  if (!role || !readBool(role.active, true)) return { authenticated: false };
  const credential = await getCredential(session.email);
 
  if (touch) {
    try {
      await wixData.update(COLLECTIONS.sessions, { ...session, lastSeenAt: new Date() }, { suppressAuth: true });
    } catch (_) {}
  }
 
  return buildProfile(user, role, session, credential);
}

/**
 * @param {{ sessionToken?: any, area?: string, action?: string, specialPermission?: string }} [options]
 */
export async function requireStaffAccess({ sessionToken, area = '', action = 'View', specialPermission = '' } = {}) {
  const state = await getSessionState({ sessionToken, touch: true });
  if (!state?.authenticated) throw new Error('AUTH_REQUIRED');
  if (state.isAdmin || text(state.roleKey).toLowerCase() === 'admin' || readBool(state.permissions?.__admin, false)) return state;
  if (specialPermission) {
    const hasSpecial = Array.isArray(state.specialPermissions) && state.specialPermissions.includes(String(specialPermission));
    if (!hasSpecial) throw new Error('ACCESS_DENIED');
  }
  if (!canAccessArea(state.permissions, area, action)) throw new Error('ACCESS_DENIED');
  return state;
}

    };
  }

      mustChangePassword: readBool(cred?.mustChangePassword, false),

  const canOverridePricing = isAdmin || hasSpecialPermission(state, 'overridePricing');
  const canApproveManualOverride = isAdmin || hasSpecialPermission(state, 'approveManualOverride');
  return {
    accessState: 'ready',
    currentStaff: {
      email: state.email,
      name: state.fullName,
      roleKey: state.roleKey,
      roleLabel: state.roleLabel,
      isAdmin,
      canManageStaff,
      canViewAudit,
      canManageBusinessSettings,
      canOverridePricing,
      canApproveManualOverride,
      permissions: currentRoleView.permissions || {},
      permissionsSummary: summarizeRole({ permissions: currentRoleView.permissions }),

      mustChangePassword: !!state.mustChangePassword,

    },

    auditRows,
    sessionRows,
    meta: {
      modules: MODULES,
      actions: ACTIONS,
      specialPermissions: SPECIALS,
      specialPermissionDescriptions: SPECIALS.map((key) => ({ key, description: SPECIAL_DESCRIPTIONS[key] || '' })),
      pendingRecoveryCount: auditRows.filter((row) => String(row.entityType || '') === 'accessRecovery' && String(row.action || '') === 'requestRecovery').length,
      note: 'Ξεχωριστός λογαριασμός ανά χρήστη. Deactivate αντί για delete, ώστε να διατηρείται το audit trail.'

export async function inviteStaffUser({ sessionToken, payload = {} } = {}) {
  const ctx = await requireStaffAccess({ sessionToken, area: '', action: 'View', specialPermission: 'manageStaff' });
  const safeEmail = normalizeEmail(payload.email);
  if (!safeEmail) throw new Error('Missing email');
  const current = await getStaffUser(safeEmail);
  const roleKey = slugKey(payload.roleKey || current?.roleKey || 'viewer');
  const role = await getRole(roleKey);
  if (!role) throw new Error('Role not found');

  const userPayload = {
    _id: current?._id,
    email: safeEmail,
    fullName: text(payload.fullName || current?.fullName || safeEmail.split('@')[0]),
    roleKey,
    active: payload.active !== undefined ? readBool(payload.active, true) : (current ? readBool(current.active, true) : true),
    memberId: text(payload.memberId || current?.memberId),
    notes: text(payload.notes || current?.notes)
  };
  const userRow = await upsertStaffUser({ sessionToken, payload: userPayload });
  const tempPassword = buildTempPassword();
  await upsertCredential(safeEmail, tempPassword, { mustChangePassword: true, active: true, notes: 'Temporary onboarding password issued by admin.' });
  const currentTokenHash = sessionToken ? sha256(sessionToken) : '';
  await revokeSessionsByQuery(wixData.query(COLLECTIONS.sessions).eq('email', safeEmail), currentTokenHash, false);
  await logAudit({
    entityType: 'user',
    entityId: safeEmail,
    action: current ? 'user.reinvite' : 'user.invite',
    summary: current ? 'Εκδόθηκε νέο προσωρινό password / re-onboard' : 'Δημιουργήθηκε νέος staff user και onboarding access',
    oldValue: current ? safeJson({ roleKey: current.roleKey, active: current.active }) : '',
    newValue: safeJson({ roleKey: userRow.roleKey, active: userRow.active }),
    actorEmail: ctx.email,
    actorName: ctx.fullName
  });
  return {
    success: true,
    mode: current ? 'reinvite' : 'invite',
    email: safeEmail,
    fullName: userRow.fullName,
    roleKey,
    roleLabel: role.label || roleKey,
    tempPassword,
    mustChangePassword: true,
    onboardingMessage: buildOnboardingMessage({
      fullName: userRow.fullName,
      email: safeEmail,
      tempPassword,
      roleLabel: role.label || roleKey
    })
  };
}

export async function revokeStaffSessions({ sessionToken, email, itemId = '', exceptCurrent = false } = {}) {
  const ctx = await requireStaffAccess({ sessionToken, area: '', action: 'View', specialPermission: 'manageStaff' });
  const currentTokenHash = sessionToken ? sha256(sessionToken) : '';
  let revoked = 0;
  if (itemId) {
    const row = await wixData.get(COLLECTIONS.sessions, itemId, { suppressAuth: true }).catch(() => null);
    if (row && readBool(row.active, true) && (!exceptCurrent || String(row.tokenHash || '') !== currentTokenHash)) {
      await wixData.update(COLLECTIONS.sessions, { ...row, active: false, revokedAt: new Date() }, { suppressAuth: true });
      revoked = 1;
    }
  } else {
    const safeEmail = normalizeEmail(email);
    if (!safeEmail) throw new Error('Missing email');
    revoked = await revokeSessionsByQuery(wixData.query(COLLECTIONS.sessions).eq('email', safeEmail), currentTokenHash, exceptCurrent);
  }
  await logAudit({ entityType: 'session', entityId: itemId || normalizeEmail(email), action: 'session.revoke', summary: `Ανακλήθηκαν ${revoked} session(s)`, actorEmail: ctx.email, actorName: ctx.fullName });
  return { success: true, revoked };
}

async function hasRecentRecoveryRequest(email, minutes = 15) {
  const safeEmail = normalizeEmail(email);
  if (!safeEmail) return false;
  try {
    const since = addHours(new Date(), -(Number(minutes || 0) / 60));
    const res = await wixData.query(COLLECTIONS.audit)
      .eq('entityType', 'accessRecovery')
      .eq('entityId', safeEmail)
      .eq('action', 'requestRecovery')
      .ge('createdAt', since)
      .descending('createdAt')
      .limit(1)
      .find({ suppressAuth: true });
    return !!(res.items && res.items.length);
  } catch (_) {
    return false;
  }
}

export async function requestAccessRecovery({ email, userAgent = '' } = {}) {
  const safeEmail = normalizeEmail(email);
  const generic = {
    success: true,
    accepted: true,
    message: 'Η αίτηση recovery καταχωρήθηκε. Αν ο λογαριασμός είναι ενεργός, ο διαχειριστής θα δει το αίτημα στο backroom και θα εκδώσει νέο προσωρινό password.'
  };
  if (!safeEmail) return { ...generic, accepted: false, message: 'Συμπλήρωσε πρώτα το εταιρικό email σου.' };
  const user = await getStaffUser(safeEmail).catch(() => null);
  if (!user || !readBool(user.active, true)) {
    return generic;
  }
  const recent = await hasRecentRecoveryRequest(safeEmail, 15);
  if (recent) {
    return { ...generic, accepted: false, message: 'Υπάρχει ήδη πρόσφατο αίτημα recovery για αυτό το email. Αν δεν έχεις απάντηση, μίλησε με τον διαχειριστή.' };
  }
  await logAudit({
    entityType: 'accessRecovery',
    entityId: safeEmail,
    action: 'requestRecovery',
    summary: `Recovery requested for ${safeEmail}`,
    actorEmail: safeEmail,
    actorName: text(user.fullName),
    newValue: JSON.stringify({ status: 'pending', userAgent: text(userAgent).slice(0, 280) })
  });
  return generic;
}

async function doPurgeExpiredSessions() {
  const now = new Date();
  const [inactiveRes, expiredRes] = await Promise.all([
    wixData.query(COLLECTIONS.sessions).eq('active', false).limit(1000).find({ suppressAuth: true }).catch(() => ({ items: [] })),
    wixData.query(COLLECTIONS.sessions).lt('expiresAt', now).eq('active', true).limit(1000).find({ suppressAuth: true }).catch(() => ({ items: [] }))
  ]);
  const ids = [...new Set([
    ...(inactiveRes.items || []).map((i) => i._id),
    ...(expiredRes.items || []).map((i) => i._id)
  ])].filter(Boolean);
  if (!ids.length) return { deleted: 0 };
  let deleted = 0;
  for (let i = 0; i < ids.length; i += 100) {
    try {
      const res = await wixData.bulkRemove(COLLECTIONS.sessions, ids.slice(i, i + 100), { suppressAuth: true });
      deleted += (res?.removed ?? ids.slice(i, i + 100).length);
    } catch (_) {}
  }
  return { deleted };
}
